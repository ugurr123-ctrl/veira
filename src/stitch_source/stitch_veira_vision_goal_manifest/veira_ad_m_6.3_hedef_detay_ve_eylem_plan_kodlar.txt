class MilestoneItem {
  final String id;
  final String title;
  final String quarter; // Q1, Q2, Q3
  final bool isCompleted;

  const MilestoneItem({
    required this.id,
    required this.title,
    required this.quarter,
    this.isCompleted = false,
  });

  MilestoneItem copyWith({
    String? id,
    String? title,
    String? quarter,
    bool? isCompleted,
  }) {
    return MilestoneItem(
      id: id ?? this.id,
      title: title ?? this.title,
      quarter: quarter ?? this.quarter,
      isCompleted: isCompleted ?? this.isCompleted,
    );
  }
}

class GoalDetailModel {
  final String id;
  final String title;
  final String category;
  final String emotionalWhy;
  final String imageUrl;
  final int daysLeft;
  final double progress; // 0.0 - 1.0
  final String priority;
  final List<MilestoneItem> milestones;
  final String todaysMicroAction;
  final int estimatedMinutes;
  final bool isActionDone;
  final bool isShrinkingAi;

  const GoalDetailModel({
    required this.id,
    required this.title,
    required this.category,
    required this.emotionalWhy,
    required this.imageUrl,
    required this.daysLeft,
    required this.progress,
    required this.priority,
    required this.milestones,
    required this.todaysMicroAction,
    this.estimatedMinutes = 15,
    this.isActionDone = false,
    this.isShrinkingAi = false,
  });

  GoalDetailModel copyWith({
    String? id,
    String? title,
    String? category,
    String? emotionalWhy,
    String? imageUrl,
    int? daysLeft,
    double? progress,
    String? priority,
    List<MilestoneItem>? milestones,
    String? todaysMicroAction,
    int? estimatedMinutes,

  }) {

      id: id ?? this.id,

      category: category ?? this.category,
      emotionalWhy: emotionalWhy ?? this.emotionalWhy,
      imageUrl: imageUrl ?? this.imageUrl,
      daysLeft: daysLeft ?? this.daysLeft,
      progress: progress ?? this.progress,
      priority: priority ?? this.priority,

## 3. RIVERPOD DURUM YÖNETİCİSİ

### Dosya Yolu: `lib/features/goals/providers/goal_detail_provider.dart`
```dart
import 'package:flutter_riverpod/flutter_riverpod.dart';
import '../domain/goal_detail_model.dart';

final goalDetailProvider = StateNotifierProvider<GoalDetailNotifier, GoalDetailModel>((ref) {
  return GoalDetailNotifier();
});

class GoalDetailNotifier extends StateNotifier<GoalDetailModel> {
  GoalDetailNotifier()
      : super(
          const GoalDetailModel(
            id: 'goal-1',
            title: 'Como Gölü Kıyısında Yaşam & Çalışma',
            category: 'Kariyer & Bolluk',
            emotionalWhy:
                'Aileme özgür ve huzurlu bir yaşam alanı sunmak, sabahları doğaya uyanıp kendi potansiyelimi en yüksek seviyede gerçekleştirmek.',
            imageUrl:
                'https://images.unsplash.com/photo-1507525428034-b723cf961d3e?w=800&q=80',
            daysLeft: 142,
            progress: 0.45,
            priority: 'Yüksek Öncelik',
            milestones: [
              MilestoneItem(
                id: 'm1',
                title: 'Q1: İtalyanca B1 Dil Sertifikası & Vize Portföyü',
                quarter: 'Q1',
                isCompleted: true,
              ),
              MilestoneItem(
                id: 'm2',
                title: 'Q2: Como/Milano Çalışma Sözleşmesi & Şirket Evrakları',
                quarter: 'Q2',
                isCompleted: false,
              ),
              MilestoneItem(
                id: 'm3',
                title: 'Q3: Ev Kiralama, Uçak Bileti ve Yerleşim',
                quarter: 'Q3',
                isCompleted: false,
              ),
            ],
            todaysMicroAction:
                'İtalyanca çalışma portföyünden 1 vize başvuru yazısını incele ve 3 anahtar kelimeyi not al.',
            estimatedMinutes: 15,
            isActionDone: false,
          ),
        );

  void toggleAction() {
    state = state.copyWith(isActionDone: !state.isActionDone);
  }

  void toggleMilestone(String milestoneId) {
    final updated = state.milestones.map((m) {
      if (m.id == milestoneId) {
        return m.copyWith(isCompleted: !m.isCompleted);
      }
      return m;
    }).toList();

    // Yeni ilerleme yüzdesini hesapla
    final completedCount = updated.where((m) => m.isCompleted).length;
    final newProgress = completedCount / updated.length;

    state = state.copyWith(
      milestones: updated,
      progress: newProgress,
    );
  }

  // Kullanıcı "Üşeniyorum / Zamanım Yok" dediğinde adımı 5 dakikaya indiren AI kuralı
  Future<void> shrinkAction() async {
    state = state.copyWith(isShrinkingAi: true);
    await Future.delayed(const Duration(milliseconds: 600));

    state = state.copyWith(
      todaysMicroAction:
          '5 Dakikalık Mikro-Adım: Sadece vize formunun ilk sayfasını aç ve başlığını oku.',
      estimatedMinutes: 5,
      isShrinkingAi: false,
    );
  }
}
```
