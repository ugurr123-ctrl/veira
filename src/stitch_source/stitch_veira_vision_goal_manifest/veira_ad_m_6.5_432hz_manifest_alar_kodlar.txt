## 2. MODELLER (SES OYNATICI VE MİKSER DURUMU)

### Dosya Yolu: `lib/features/manifest/domain/manifest_player_model.dart`
```dart
class ManifestPlayerModel {
  final bool isPlaying;
  final double frequencyVolume; // 0.0 - 1.0 (432Hz Tonu)
  final double ambientVolume;   // 0.0 - 1.0 (Gece Yağmuru)
  final int selectedTimerMinutes; // 0 (Kapalı), 15, 30, 45, 60
  final Duration currentPosition;
  final Duration totalDuration;
  final String activeFrequency; // '432Hz' veya '528Hz'

  const ManifestPlayerModel({
    this.isPlaying = false,
    this.frequencyVolume = 0.85,
    this.ambientVolume = 0.40,
    this.selectedTimerMinutes = 15,
    this.currentPosition = const Duration(minutes: 4, seconds: 12),
    this.totalDuration = const Duration(minutes: 15),
    this.activeFrequency = '432Hz',
  });

  ManifestPlayerModel copyWith({
    bool? isPlaying,
    double? frequencyVolume,
    double? ambientVolume,
    int? selectedTimerMinutes,
    Duration? currentPosition,
    Duration? totalDuration,
    String? activeFrequency,
  }) {
    return ManifestPlayerModel(
      isPlaying: isPlaying ?? this.isPlaying,
      frequencyVolume: frequencyVolume ?? this.frequencyVolume,
      ambientVolume: ambientVolume ?? this.ambientVolume,
      selectedTimerMinutes: selectedTimerMinutes ?? this.selectedTimerMinutes,
      currentPosition: currentPosition ?? this.currentPosition,
      totalDuration: totalDuration ?? this.totalDuration,
      activeFrequency: activeFrequency ?? this.activeFrequency,
    );
  }
}
```

## 3. RIVERPOD DURUM YÖNETİCİSİ (DUAL AUDIO NOTIFIER)

### Dosya Yolu: `lib/features/manifest/providers/manifest_player_provider.dart`
```dart
import 'dart:async';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import '../domain/manifest_player_model.dart';

final manifestPlayerProvider =
    StateNotifierProvider<ManifestPlayerNotifier, ManifestPlayerModel>((ref) {
  return ManifestPlayerNotifier();
});

class ManifestPlayerNotifier extends StateNotifier<ManifestPlayerModel> {
  Timer? _positionTimer;

  ManifestPlayerNotifier() : super(const ManifestPlayerModel()) {
    // Demo simülasyonu için başlat
    _startTimer();
  }

  void _startTimer() {
    _positionTimer = Timer.periodic(const Duration(seconds: 1), (timer) {
      if (state.isPlaying) {
        if (state.currentPosition < state.totalDuration) {
          state = state.copyWith(
            currentPosition: state.currentPosition + const Duration(seconds: 1),
          );
        } else {
          state = state.copyWith(
            currentPosition: Duration.zero,
            isPlaying: false,
          );
        }
      }
    });
  }

  void togglePlayPause() {
    state = state.copyWith(isPlaying: !state.isPlaying);
  }

  void setFrequencyVolume(double val) {
    state = state.copyWith(frequencyVolume: val);
  }

  void setAmbientVolume(double val) {
    state = state.copyWith(ambientVolume: val);
  }

  void setFrequency(String freq) {
    state = state.copyWith(activeFrequency: freq);
  }

  void setTimerMinutes(int minutes) {
    state = state.copyWith(selectedTimerMinutes: minutes);
  }

  void seekTo(Duration position) {
    state = state.copyWith(currentPosition: position);
  }

  void seekForward10() {
    final next = state.currentPosition + const Duration(seconds: 10);
    state = state.copyWith(
      currentPosition: next > state.totalDuration ? state.totalDuration : next,
    );
  }

  void seekBackward10() {
    final prev = state.currentPosition - const Duration(seconds: 10);
    state = state.copyWith(
      currentPosition: prev < Duration.zero ? Duration.zero : prev,
    );
  }

  @override
  void dispose() {
    _positionTimer?.cancel();
    super.dispose();
  }
}
```
