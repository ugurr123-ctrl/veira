## 2. MODELLER (BOARD ITEM & CANVAS CONFIG)

### Dosya Yolu: `lib/features/vision_board/domain/board_item_model.dart`
```dart
class BoardItemModel {
  final String id;
  final String imageUrl;
  final String title;
  final String category;
  final String? affirmationText;
  final String? connectedGoalId;
  final String? connectedMicroAction;
  final double posX;
  final double posY;
  final double scale;
  final double rotation; // radyan cinsinden
  final int layerOrder;
  final bool isCenterpiece; // 1 Numaralı Ana Niyet Yıldızı

  const BoardItemModel({
    required this.id,
    required this.imageUrl,
    required this.title,
    required this.category,
    this.affirmationText,
    this.connectedGoalId,
    this.connectedMicroAction,
    this.posX = 0.0,
    this.posY = 0.0,
    this.scale = 1.0,
    this.rotation = 0.0,
    this.layerOrder = 0,
    this.isCenterpiece = false,
  });

  BoardItemModel copyWith({
    String? id,
    String? imageUrl,
    String? title,
    String? category,
    String? affirmationText,
    String? connectedGoalId,
    String? connectedMicroAction,
    double? posX,
    double? posY,
    double? scale,
    double? rotation,
    int? layerOrder,
    bool? isCenterpiece,
  }) {
    return BoardItemModel(
      id: id ?? this.id,
      imageUrl: imageUrl ?? this.imageUrl,
      title: title ?? this.title,
      category: category ?? this.category,
      affirmationText: affirmationText ?? this.affirmationText,
      connectedGoalId: connectedGoalId ?? this.connectedGoalId,
      connectedMicroAction: connectedMicroAction ?? this.connectedMicroAction,
      posX: posX ?? this.posX,
      posY: posY ?? this.posY,
      scale: scale ?? this.scale,
      rotation: rotation ?? this.rotation,
      layerOrder: layerOrder ?? this.layerOrder,
      isCenterpiece: isCenterpiece ?? this.isCenterpiece,
    );
  }
}
```

import 'package:flutter_riverpod/flutter_riverpod.dart';
import '../domain/board_item_model.dart';

class VisionBoardState {
  final List<BoardItemModel> items;
  final String? selectedItemId;
  final bool isExporting;
  final String activeTemplate; // 'polaroid', 'magazine', 'luxury_dark'

  const VisionBoardState({
    required this.items,
    this.selectedItemId,
    this.isExporting = false,
    this.activeTemplate = 'luxury_dark',
  });

  int get count => items.length;
  bool get isMaxReached => items.length >= 12;

  VisionBoardState copyWith({
    List<BoardItemModel>? items,
    String? selectedItemId,
    bool? isExporting,
    String? activeTemplate,
  }) {
    return VisionBoardState(
      items: items ?? this.items,
      selectedItemId: selectedItemId,
      isExporting: isExporting ?? this.isExporting,
      activeTemplate: activeTemplate ?? this.activeTemplate,
    );
  }
}

final visionBoardProvider = StateNotifierProvider<VisionBoardNotifier, VisionBoardState>((ref) {

  void selectItem(String? id) {
    state = state.copyWith(selectedItemId: id);
  }

  void updatePosition(String id, double dx, double dy) {
    final updated = state.items.map((item) {
      if (item.id == id) {
        return item.copyWith(posX: item.posX + dx, posY: item.posY + dy);
      }
      return item;
    }).toList();
    state = state.copyWith(items: updated);
  }

  void updateScaleAndRotation(String id, double scaleDelta, double rotationDelta) {
    final updated = state.items.map((item) {
      if (item.id == id) {
        return item.copyWith(
          scale: (item.scale * scaleDelta).clamp(0.6, 2.5),
          rotation: item.rotation + rotationDelta,
        );
      }
      return item;
    }).toList();
    state = state.copyWith(items: updated);
  }

  Future<void> exportHdWallpaper() async {
    state = state.copyWith(isExporting: true);
    await Future.delayed(const Duration(milliseconds: 1400));
    state = state.copyWith(isExporting: false);
  }
}
