class DailyRoutineModel {
  final String userName;
  final int streakDays;
  final String dailyAffirmation;
  final String connectedGoalTitle;
  final String connectedGoalCategory;
  final String microActionText;
  final int estimatedMinutes;
  final bool isActionCompleted;
  final String goalImageUrl;
  final Map<String, double> categoryProgress; // 'career': 0.75, 'health': 0.40, 'spirit': 0.85

  const DailyRoutineModel({
    required this.userName,
    required this.streakDays,
    required this.dailyAffirmation,
    required this.connectedGoalTitle,
    required this.connectedGoalCategory,
    required this.microActionText,
    this.estimatedMinutes = 15,
    this.isActionCompleted = false,
    required this.goalImageUrl,
    required this.categoryProgress,
  });

  DailyRoutineModel copyWith({
    String? userName,
    int? streakDays,
    String? dailyAffirmation,
    String? connectedGoalTitle,
    String? connectedGoalCategory,
    String? microActionText,
    int? estimatedMinutes,
    bool? isActionCompleted,
    String? goalImageUrl,
    Map<String, double>? categoryProgress,
  }) {
    return DailyRoutineModel(
      userName: userName ?? this.userName,
      streakDays: streakDays ?? this.streakDays,
      dailyAffirmation: dailyAffirmation ?? this.dailyAffirmation,
      connectedGoalTitle: connectedGoalTitle ?? this.connectedGoalTitle,
      connectedGoalCategory: connectedGoalCategory ?? this.connectedGoalCategory,
      microActionText: microActionText ?? this.microActionText,
      estimatedMinutes: estimatedMinutes ?? this.estimatedMinutes,
      isActionCompleted: isActionCompleted ?? this.isActionCompleted,
      goalImageUrl: goalImageUrl ?? this.goalImageUrl,
      categoryProgress: categoryProgress ?? this.categoryProgress,
    );
  }
}

import 'package:flutter_riverpod/flutter_riverpod.dart';

final dailyRoutineProvider = StateNotifierProvider<DailyRoutineNotifier, DailyRoutineModel>((ref) {
  return DailyRoutineNotifier();
});

class DailyRoutineNotifier extends StateNotifier<DailyRoutineModel> {
  DailyRoutineNotifier()
      : super(
          const DailyRoutineModel(
            userName: 'Selin',
            streakDays: 14,
            dailyAffirmation:
                'Bugün attığım her küçük adım, 2027 vizyonumun bir parçası. Dinginlik ve güç içimde.',
            connectedGoalTitle: 'Como Gölü Kıyısında Yaşam & Çalışma',
            connectedGoalCategory: 'career',
            microActionText: 'İtalyanca çalışma portföyünden 1 vize yazısını incele ve 3 not al.',
            estimatedMinutes: 15,
            isActionCompleted: false,
            goalImageUrl:
                'https://images.unsplash.com/photo-1507525428034-b723cf961d3e?w=800&q=80',
            categoryProgress: {
              'career': 0.75,
              'health': 0.45,
              'spirit': 0.85,
            },
          ),
        );

  void toggleMicroAction() {
    final nextState = !state.isActionCompleted;
    state = state.copyWith(
      isActionCompleted: nextState,
      streakDays: nextState ? state.streakDays + 1 : (state.streakDays > 0 ? state.streakDays - 1 : 0),
    );
  }

  void updateAffirmation(String newAffirmation) {
    state = state.copyWith(dailyAffirmation: newAffirmation);
  }
}

## 4. ÖZEL GRAFİK BİLEŞENİ: 3 İÇ İÇE GEÇMİŞ İLERLEME HALKASI

### Dosya Yolu: `lib/features/home/presentation/widgets/concentric_progress_rings.dart`
```dart
import 'dart:math';
import 'package:flutter/material.dart';
import '../../../../core/theme/app_colors.dart';

class ConcentricProgressRings extends StatelessWidget {
  final double careerProgress;
  final double healthProgress;
  final double spiritProgress;
  final double size;

  const ConcentricProgressRings({
    super.key,
    required this.careerProgress,
    required this.healthProgress,
    required this.spiritProgress,
    this.size = 110,
  });

  @override
  Widget build(BuildContext context) {
    return SizedBox(
      width: size,
      height: size,
      child: CustomPaint(
        painter: _ConcentricRingsPainter(
          career: careerProgress,
          health: healthProgress,
          spirit: spiritProgress,
        ),
      ),
    );
  }
}

class _ConcentricRingsPainter extends CustomPainter {
  final double career;
  final double health;
  final double spirit;

  _ConcentricRingsPainter({
    required this.career,
    required this.health,
    required this.spirit,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final center = Offset(size.width / 2, size.height / 2);
    final baseStrokeWidth = 7.0;

    // Dış Halka: Kariyer (Celestial Gold)
    _drawRing(
      canvas,
      center,
      radius: (size.width / 2) - 6,
      progress: career,
      strokeColor: AppColors.celestialGold,
      bgColor: AppColors.celestialGold.withOpacity(0.15),
      strokeWidth: baseStrokeWidth,
    );

    // Orta Halka: Sağlık (Ethereal Teal)
    _drawRing(
      canvas,
      center,
      radius: (size.width / 2) - 18,
      progress: health,
      strokeColor: AppColors.etherealTeal,
      bgColor: AppColors.etherealTeal.withOpacity(0.15),
      strokeWidth: baseStrokeWidth,
    );

    // İç Halka: Ruhsal Denge (Mystic Violet)
    _drawRing(
      canvas,
      center,
      radius: (size.width / 2) - 30,
      progress: spirit,
      strokeColor: AppColors.mysticViolet,
      bgColor: AppColors.mysticViolet.withOpacity(0.15),
      strokeWidth: baseStrokeWidth,
    );
  }

  void _drawRing(
    Canvas canvas,
    Offset center, {
    required double radius,
    required double progress,
    required Color strokeColor,
    required Color bgColor,
    required double strokeWidth,
  }) {
    final bgPaint = Paint()
      ..color = bgColor
      ..style = PaintingStyle.stroke
      ..strokeWidth = strokeWidth
      ..strokeCap = StrokeCap.round;

    final fgPaint = Paint()
      ..color = strokeColor
      ..style = PaintingStyle.stroke
      ..strokeWidth = strokeWidth
      ..strokeCap = StrokeCap.round;

    // Arka plan tam çember
    canvas.drawCircle(center, radius, bgPaint);

    // İlerleme Yayı (-pi/2'den başlayarak saat yönünde)
    final sweepAngle = 2 * pi * progress.clamp(0.0, 1.0);
    canvas.drawArc(
      Rect.fromCircle(center: center, radius: radius),
      -pi / 2,
      sweepAngle,
      false,
      fgPaint,
    );
  }

  @override
  bool shouldRepaint(covariant _ConcentricRingsPainter oldDelegate) {
    return oldDelegate.career != career ||
        oldDelegate.health != health ||
        oldDelegate.spirit != spirit;
  }
}
```
